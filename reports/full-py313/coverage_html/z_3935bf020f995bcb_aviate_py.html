33class Aviate(Function): 

34 """ 

35 Function that moves the drone within the environment. 

36 

37 Examples 

38 -------- 

39 >>> t = Trajectories() 

40 >>> t.s.put(z=100.0, goal_z=100.0, dz=0.0, dx=10.0) 

41 >>> des_traj = t.create_local("des_traj") 

42 >>> av = Aviate(trajectories=t) 

43 >>> av.static_behavior() 

44 >>> av.dynamic_behavior() 

45 >>> av.trajectories.s.x 

46 10.0 

47 >>> av.trajectories.des_traj.s.put(dx=0.0, dy=10.0) 

48 >>> av.static_behavior() 

49 >>> av.dynamic_behavior() 

50 >>> av.trajectories.s.y 

51 10.0 

52 """ 

53 

54 __slots__ = ('trajectories', 'force', 'electricity', 'environment') 

55 

56 flow_trajectories = Trajectories 

57 flow_force = Force 

58 flow_electricity = Electricity 

59 flow_environment = AircraftEnvironment 

60 container_m = AviateMode

62 def static_behavior(self): 

63 """Overall static behavior of drone (determines modes based on states).""" 

64 if self.force.s.contact_support >= 10.0: 

65 self.m.add_fault('crash') 

66 

67 if self.trajectories.s.z > 0.0: 

68 if self.electricity.s.voltage_high <= 0.0: 

69 self.m.set_mode('falling') 

70 else: 

71 self.m.set_mode('flight') 

72 else: 

73 if self.electricity.s.voltage_high > 0.0: 

74 if not self.trajectories.des_traj.s.same(dx=0.0, dy=0.0): 

75 self.m.add_fault('crash') 

76 self.m.set_mode('idle') 

77 elif not self.trajectories.des_traj.s.dz <= 0.0: 

78 self.m.set_mode('flight') 

79 else: 

80 self.m.set_mode('idle') 

81 else: 

82 self.m.set_mode('idle') 

83 

84 if self.m.in_mode("falling", "crash"): 

85 self.falling_behavior() 

86 elif self.m.in_mode('flight'): 

87 self.flight_behavior() 

88 elif self.m.in_mode("idle"): 

89 self.idle_behavior() 

90 

91 def dynamic_behavior(self): 

92 """Overall dynamic behavior of the drone (flight, falling, and idle).""" 

93 self.trajectories.s.increment_position() 

94 self.trajectories.s.limit(z=(0.0, np.inf)) 

95 

96 def flight_behavior(self): 

97 """ 

98 Behavior when the drone is flying in the air. 

99 

100 The trajectory increments to a new location determined by the direction and 

101 distance of the desired trajectory. 

102 """ 

103 self.trajectories.s.assign(self.trajectories.des_traj.s, 'dx', 'dy', 'dz') 

104 self.electricity.s.current_high = 1.0 

105 self.force.s.put(lift_support=1.0) 

106 

107 def falling_behavior(self): 

108 """ 

109 Behavior when falling. 

110 

111 The drone falls to the ground, removing all support to the drone. 

112 """ 

113 if self.trajectories.s.z > 0.0: 

114 #self.trajectories.s.assign(self.trajectories.des_traj.s, 'dx', 'dy') 

115 self.trajectories.s.dz = -self.trajectories.s.z 

116 self.force.s.put(lift_support=0.0) 

117 else: 

118 self.idle_behavior() 

119 

120 def idle_behavior(self): 

121 """Behavior when not moving and grounded.""" 

122 self.trajectories.s.put(dx=0.0, dy=0.0, dz=0.0) 

123 self.force.s.put(lift_support=0.0) 

124 self.electricity.s.current_high = 0.0

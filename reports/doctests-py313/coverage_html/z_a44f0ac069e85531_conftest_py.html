1# -*- coding: utf-8 -*- 

2""" 

3Pytest configuration file. 

4 

5Adds options to pytest: 

6 - testtype: str 

7 Defines type of tests to run. May be 

8 "full" if all tests 

9 "doctests" for just doctests 

10 "notebooks-fast"/"notebooks-slow"/"notebooks" for different lists of notebooks 

11 or any string for a custom test type 

12 - auto_build_reports : bool 

13 Whether to build a report given the testtype options. Default is False. 

14 If True, reports will go in the reports/testtype-pyver folder 

15 

16Examples 

17-------- 

18pytest --testtype=fast-notebooks # should run all fast notebooks 

19pytest --testtype=doctests # should run all doctest modules 

20pytest --testtype=doctests 

21# the below command should test one doctest file while  

22# note that cov_report "html:auto" is required for a coverage report to be generated 

23# while 

24pytest --doctest-modules fmdtools/define/container/base.py --testtype=custom --auto_build_reports=True --cov-report "html:auto" 

25# typically, for testing, we just use this for fast tests: 

26pytest --testtype="doctest" -auto_build_reports=True --cov-report "html:auto" 

27# and this for comprehensive tests: 

28pytest --testtype="full" --auto_build_reports=True --cov-report "html:auto"

136def pytest_addoption(parser): 

137 """ 

138 Add option for testtype, pyver, and auto_build_reports to pytest. 

139 

140 See pytest API for details on usage. 

141 """ 

142 parser.addoption("--skiplist", action="store_true", 

143 default=too_slow_notebooks, help="skip listed tests") 

144 parser.addoption("--testtype", action="store", default="doctests", 

145 help="test type: full, doctests, or any other name", 

146 type=str) 

147 parser.addoption("--auto_build_reports", action="store", default=False, 

148 help="build_report: whether to build a report", 

149 type=bool)

152def pytest_collection_modifyitems(config, items): 

153 """Skip listed (too slow notebooks) by default.""" 

154 tests_to_skip = config.getoption("--skiplist") 

155 if not tests_to_skip: 

156 # --skiplist not given in cli, therefore move on 

157 return 

158 skip_listed = pytest.mark.skip(reason="included in --skiplist") 

159 for item in items: 

160 for testpath in tests_to_skip: 

161 if Path(testpath).samefile(item.path): 

162 item.add_marker(skip_listed)

165def pytest_configure(config): 

166 """ 

167 Configure pytest given testtype and auto_build_reports options. 

168 

169 If testtype is given, only runs the appropriate tests for that type. Also creates 

170 paths for html test and coverage reports given auto_build_reports option. 

171 """ 

172 plt.rcParams['figure.max_open_warning'] = 50 

173 

174 testtype = config.getoption("--testtype") 

175 pyver = "py"+str(sys.version_info.major)+str(sys.version_info.minor) 

176 reportdir = "./reports/"+testtype+"-"+pyver 

177 if config.getoption("--auto_build_reports") and 'cov_report' in config.option: 

178 if not config.option.cov_source: 

179 raise Exception("Coverage report will not build without --cov option.") 

180 config.option.cov_report['html'] = reportdir+"/coverage_html" 

181 config.option.cov_report['xml'] = reportdir+"/coverage/coverage.xml" 

182 

183 config.option.htmlpath = reportdir+"/junit/report.html" 

184 config.option.xmlpath = reportdir+"/junit/junit.xml" 

185 if "full" not in testtype and "custom" not in testtype: 

186 config.args = [] 

187 if "doctests" in testtype: 

188 config.doctestmodules=True 

189 if "custom" not in testtype: 

190 config.option.file_or_dir = str(doctest_modules) 

191 config.args.extend(doctest_modules) 

192 

193 if "notebooks" in testtype: 

194 if "slow" in testtype: 

195 config.args.extend(slow_notebooks) 

196 elif "fast" in testtype: 

197 config.args.extend(fast_notebooks) 

198 else: 

199 config.args.extend(fast_notebooks+slow_notebooks)
